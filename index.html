// Registrar el Service Worker para funcionamiento offline if ('serviceWorker' in navigator) { navigator.serviceWorker.register('sw.js'); } let entrenamientos = JSON.parse(localStorage.getItem('gymData')) || []; mostrarHistorial(); function guardarRegistro() { const ejercicio = document.getElementById('ejercicio').value; const peso = document.getElementById('peso').value; const repes = document.getElementById('repes').value; if(!ejercicio || !peso) return alert('Por favor, indica ejercicio y peso.'); const ahora = new Date(); const nuevoRegistro = { ejercicio, peso, repes, dia: ahora.toLocaleDateString(), // Guarda la fecha (Ej: 05/10/2026) hora: ahora.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) // Guarda la hora (Ej: 19:30) }; entrenamientos.push(nuevoRegistro); localStorage.setItem('gymData', JSON.stringify(entrenamientos)); // Limpiar solo peso y repes para la siguiente serie document.getElementById('peso').value = ''; document.getElementById('repes').value = ''; mostrarHistorial(); } function mostrarHistorial() { const contenedor = document.getElementById('historial'); contenedor.innerHTML = ''; // 1. Agrupar los registros por día const gruposPorDia = {}; entrenamientos.forEach(ent => { // Retrocompatibilidad por si ya habías guardado datos con el código anterior const dia = ent.dia || ent.fecha.split(',')[0]; if (!gruposPorDia[dia]) { gruposPorDia[dia] = []; } gruposPorDia[dia].push(ent); }); // 2. Ordenar los días (del más reciente al más antiguo) const diasOrdenados = Object.keys(gruposPorDia).reverse(); // 3. Dibujar en la pantalla diasOrdenados.forEach(dia => { // Cabecera del día let htmlDia = `
📅 ${dia}
`; // Series de ese día (invertidas para ver la última arriba) gruposPorDia[dia].slice().reverse().forEach(ent => { const hora = ent.hora ? ent.hora : ''; htmlDia += `
${ent.ejercicio}: ${ent.peso}kg x ${ent.repes} reps ${hora}
`; }); contenedor.innerHTML += htmlDia; }); }
